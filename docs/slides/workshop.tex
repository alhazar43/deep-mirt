% ===========================================================================
\begin{frame}{Setup: amortized neural IRT}
\begin{itemize}
\item Encoder (LSTM): response history $\to$ time-varying ability $\theta_t$
\item Item embeddings $\to$ discrimination $\alpha_i>0$, difficulty $\beta_i$; GPCM decoder
\end{itemize}
\[
  P(Y_i=k\mid\theta)=\frac{\exp\sum_{m=1}^{k}\alpha_i(\theta-\beta_{i,m})}
       {\sum_{l=0}^{K-1}\exp\sum_{m=1}^{l}\alpha_i(\theta-\beta_{i,m})}
  \qquad(\text{2PL at }K{=}2)
\]
\begin{itemize}
\item Trained on response prediction only. \textbf{No supervision} on $\theta,\alpha,\beta$, no priors.
\item Scale is gauge-free ($\theta\!\to\!s\theta{+}t,\ \alpha\!\to\!\alpha/s$) $\Rightarrow$ recovery $=$ rank (Spearman $\rho$).
\end{itemize}
\end{frame}

% ===========================================================================
\begin{frame}{Architecture: shared vs decoupled item code}
The item code enters twice: the encoder ($\to\theta$) and the $\alpha,\beta$ readouts.
\medskip
\begin{itemize}
\item \textbf{Shared}: one table, width $W$ \quad(\texttt{item\_key\_dim=None})
\item \textbf{Decoupled}: \texttt{emb\_dim=8} $+$ separate \texttt{item\_key\_dim=}$W{-}8$ feeding only $\alpha,\beta$
\item \texttt{state\_alpha=True} in both $\Rightarrow$ the study varies the \emph{representation}, not the readout
\end{itemize}
\medskip
\centering
\footnotesize
shared: $\theta \leftarrow e_q \rightarrow (\alpha,\beta)$ \qquad
decoupled: $\theta\leftarrow e_q^{\text{val}}$, \ \ $e_q^{\text{key}}\rightarrow(\alpha,\beta)$
\end{frame}

\begin{frame}{Discrimination is the low-information parameter}
\begin{columns}
\begin{column}{0.5\textwidth}
\includegraphics[width=\linewidth]{s1_three_speeds.png}
\end{column}
\begin{column}{0.48\textwidth}
Gradients ($r=p-y$):
\[ \partial_\theta L=r\alpha,\ \ \partial_\beta L=-r\alpha,\ \ \partial_\alpha L=r(\theta{-}\beta) \]
Fisher ($w=p(1{-}p)$):
\[ I(\theta)=I(\beta)=\alpha^2 w,\quad I(\alpha)=(\theta{-}\beta)^2 w \]
\begin{itemize}
\item $I(\alpha)\to 0$ at $\theta{=}\beta$ (where responses concentrate)
\item stiffness $\kappa=I(\theta)/I(\alpha)$ grows with $K$
\end{itemize}
\end{column}

% ===========================================================================
\begin{frame}{Why is discrimination degraded under sharing?}
Three explanations, one test each.
\medskip
\begin{enumerate}
\item \textbf{Capacity} \,$\to$\, a wider shared code would fix it
\item \textbf{Expressivity} \,$\to$\, no shared code holds both directions at the optimum
\item \textbf{Dynamics} \,$\to$\, gradient flow drives the shared code to the $\theta$ compromise
\end{enumerate}
\medskip
Synthetic GPCM, known $\theta,\alpha,\beta$, LSTM. Each finding eliminates one.
\end{frame}

\begin{frame}{The dynamics as a gradient-flow ODE}
Item-code block near the optimum, $\delta=e_q-e_q^*$:
\[
  \dot\delta = -H\,\delta,\qquad \delta_c(t)=\delta_c(0)\,e^{-\lambda_c t}.
\]
\begin{itemize}
\item Shared block, two curvatures: $\ \kappa=\dfrac{\lambda_\theta}{\lambda_\alpha}=\dfrac{I(\theta)}{I(\alpha)}\gg1$
\item One step size $\eta<2/\lambda_{\max}$: slow mode contracts at $\eta\lambda_\alpha\sim 1/\kappa\ \Rightarrow\ \tau_\alpha^{\text{sh}}\sim O(\kappa)$
\item Decoupled block: $O(1)$-conditioned $\ \Rightarrow\ \tau_\alpha^{\text{dec}}\sim O(1)$
\end{itemize}
\[
  \boxed{\ \text{speedup}\ \sim\ \kappa = I(\theta)/I(\alpha)\ \ (\text{grows with }K)\ }
\]
Same fixed point (free-table invariant) $\Rightarrow$ \textbf{rate, not endpoint}. Largest under
plain GD; Adam's preconditioner compresses $\kappa$.
\end{frame}

% ===========================================================================
\begin{frame}{Conclusion}
\begin{block}{}
Decoupling the discrimination code buys an $O(\kappa)$ convergence-rate advantage
on discrimination rank, $\kappa=I(\theta)/I(\alpha)$, growing with $K$, no endpoint cost.
\end{block}
\medskip
\begin{itemize}
\item not capacity (matched params), not expressivity (reached then left), but \textbf{dynamics}
\item a measurement-validity condition: recovered $\alpha$ is otherwise a finite-budget allocation artifact
\item positivity map ruled out as the lever
\end{itemize}
\medskip
\footnotesize Scope: synthetic GPCM, LSTM (architecture-independent on swap), 3--10 seeds.
Open: $\theta$/threshold channels, NRM slopes, GRM, real-data stability.
\end{frame}
