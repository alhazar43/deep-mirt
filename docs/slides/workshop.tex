% ===========================================================================
\begin{frame}{Setup: amortized neural IRT}
\begin{itemize}
\item Encoder (LSTM): response history $\to$ ability $\theta_t$
\item Item embedding $\to$ discrimination $\alpha_i$, difficulty (step thresholds) $\beta_i$; GPCM decoder
\end{itemize}
\[
  P(Y_i=k\mid\theta)=\frac{\exp\sum_{m=1}^{k}\alpha_i(\theta-\beta_{i,m})}
       {\sum_{l=0}^{K-1}\exp\sum_{m=1}^{l}\alpha_i(\theta-\beta_{i,m})}
  \qquad(\text{2PL at }K{=}2)
\]
\begin{itemize}
\item Trained on response prediction only. No supervision on $\theta,\alpha,\beta$.
\item Scale is gauge-free $\Rightarrow$ recovery $=$ rank (Spearman $\rho$).
\end{itemize}
\end{frame}

% ===========================================================================
\begin{frame}{The item code has two consumers}
\begin{itemize}
\item a \textbf{value} embedding $v_q$ (narrow) $\to$ encoder input $\to$ \textbf{ability} $\theta$
\item a \textbf{key} embedding $k_q$ (wide) $\to$ \textbf{discrimination} $\alpha$, \textbf{difficulty} $\beta$ readouts
\end{itemize}
\medskip
\begin{itemize}
\item \textbf{Shared}: one embedding, width $W$, feeds both consumers
\item \textbf{Decoupled}: a narrow value embedding $+$ a separate wide key embedding
\end{itemize}
\medskip
\centering
$\Rightarrow$ \emph{shared ties the encoder-input width to the readout width.} That tie is the problem.
\end{frame}

% ===========================================================================
\begin{frame}{First the math: the gradient flow of all three}
Per-parameter gradients, residual $r=p-y$:
\[ \partial_\theta L=r\alpha,\qquad \partial_\beta L=-r\alpha,\qquad \partial_\alpha L=r(\theta-\beta). \]
Near the optimum each parameter's mode decays at its Fisher rate ($w=p(1{-}p)$):
\[ \underbrace{I(\theta)=I(\beta)=\alpha^2 w}_{\text{fast}},\qquad
   \underbrace{I(\alpha)=(\theta-\beta)^2 w}_{\text{slow},\ \to0\ \text{at}\ \theta=\beta}. \]
\centering
Discrimination is the one low-rate mode: its lever arm $(\theta-\beta)$ vanishes exactly where
responses concentrate. \textbf{The math singles out $\alpha$ before any experiment.}
\end{frame}

% ===========================================================================
\begin{frame}{What the math predicts (then we test each)}
\begin{enumerate}
\item \textbf{Discrimination recovers last} \,(it is the single slow mode; $\theta,\beta$ fast).
\item \textbf{A shared code forces a three-way trade-off}: $\theta$ wants it narrow,
$\alpha$ wants it wide, $\beta$ is indifferent.
\item \textbf{Sharing adds conditioning} $\kappa=I(\theta)/I(\alpha)$: it throttles $\alpha$'s
rate; decoupling removes it (speedup $\sim\kappa$, growing with $K$).
\item \textbf{Positivity is necessary, the map shape is not}: among smooth positive maps the
convergence is the same; $\exp$ is not special.
\end{enumerate}
\medskip
\centering
Synthetic GPCM, known $\theta,\alpha,\beta$, LSTM. Each prediction is tested next.
\end{frame}

% ===========================================================================
\begin{frame}{The two pressures are of different KIND}
\begin{itemize}
\item \textbf{$\alpha$: a RATE effect, endpoint invariant.} $\alpha$ is a free per-item
table, so the optimum is unbiased (free-table invariant). Sharing only slows it
(conditioning $\kappa$); decoupling speeds it. Transient: vanishes as training $\to\infty$, not at a fixed budget. \hfill(P9)
\item \textbf{$\theta$: an ENDPOINT effect, not invariant.} $\theta$ is read by the
amortized encoder, NOT a free table. A wide input overfits it,
$\ \mathrm{Var}(\hat\theta)\sim \sigma^2\, W/n$. It does NOT vanish at convergence. \hfill(P10)
\item \textbf{$\beta$: indifferent control.} High Fisher, non-pooled $\Rightarrow$
neither needs width nor is hurt ($\Delta_\beta\approx0$). \hfill(P11)
\end{itemize}
\medskip
\centering
A slow mode (rate) \emph{and} an overfit (endpoint), tied by one width.
\end{frame}

\begin{frame}{Prediction 3: the gradient-flow ODE}
One item's embedding $e_q$. Near its optimum, $\delta=e_q-e_q^*$, $H_q$ the Hessian in $e_q$'s directions:
\[ \dot\delta=-H_q\delta,\qquad \delta_c(t)=\delta_c(0)e^{-\lambda_c t}. \]
\begin{itemize}
\item Shared: $e_q$ feeds encoder $(\theta)$ and $\alpha,\beta$ readouts $\Rightarrow$ steep $\theta$ direction $(\sim I(\theta))$, flat $\alpha$ direction $(\sim I(\alpha))$: $\ \kappa=I(\theta)/I(\alpha)\gg1$
\item one step $\eta<2/\lambda_{\max}$: slow mode $\sim1/\kappa\Rightarrow\tau_\alpha^{\text{sh}}\sim O(\kappa)$; decoupled $O(1)$
\end{itemize}
\[ \boxed{\ \text{speedup}\sim\kappa=I(\theta)/I(\alpha)\ (\text{grows with }K)\ } \]
Same fixed point $\Rightarrow$ rate, not endpoint. Largest under GD; Adam compresses $\kappa$.
\end{frame}

% ===========================================================================
\begin{frame}{The resolution: untie the two widths}
\begin{itemize}
\item \textbf{Shared}: one width, opposed knobs ($d\alpha/dW>0$, $d\theta/dW<0$, $d\beta/dW\approx0$) $\Rightarrow$ a Pareto frontier
\item \textbf{Decoupled}: two independent widths $\Rightarrow$ narrow encoder (good $\theta,\beta$) $+$ wide readout (good $\alpha$)
\end{itemize}
\medskip
\centering
$\Rightarrow$ decoupling is Pareto-\textbf{dominant}: it escapes the frontier, not moves along it. (P12)
\medskip

It cures both pressures at once: $\theta$ keeps its bottleneck, $\alpha$ gets its capacity.
\end{frame}

% ===========================================================================
\begin{frame}{Conclusion}
\begin{block}{}
The gradient flow predicts, and the experiments confirm, that a shared item code
forces a three-way trade-off: $\theta$ overfits wide, $\alpha$ needs wide, $\beta$
is content-narrow. Decoupling unties the widths and dominates.
\end{block}
\medskip
\begin{itemize}\small
\item Two mechanisms of different kind: $\alpha$ a convergence-RATE effect (endpoint invariant, $O(\kappa)$, grows with $K$); $\theta$ an ENDPOINT overfit ($\mathrm{Var}\sim W/n$). $\beta$ indifferent.
\item A measurement-validity condition: recovered parameters are otherwise representation-allocation artifacts.
\item Positivity map ruled out as the lever.
\end{itemize}
\medskip
\footnotesize Scope: synthetic GPCM, LSTM (architecture-independent on swap), 3 to 10 seeds.
$\theta$-overfit formalized for a linear amortizer (LSTM lift argued). Open: NRM slopes, GRM, real data.
\end{frame}
